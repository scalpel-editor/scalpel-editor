#[[
	Interface library for the Wayland backend, including source files and linking requirements.
]]

add_library(owui_backend_common_headers INTERFACE)
target_include_directories(owui_backend_common_headers INTERFACE "${CMAKE_CURRENT_LIST_DIR}")
target_sources(owui_backend_common_headers INTERFACE
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Backend.h"
)

set(OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/WaylandProtocols")
set(OWUI_XDG_SHELL_CLIENT_HEADER "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/xdg-shell-client-protocol.h")
set(OWUI_XDG_SHELL_PRIVATE_CODE "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/xdg-shell-protocol.c")
set(OWUI_XDG_DECORATION_CLIENT_HEADER "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/xdg-decoration-client-protocol.h")
set(OWUI_XDG_DECORATION_PRIVATE_CODE "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/xdg-decoration-protocol.c")
set(OWUI_PRESENTATION_TIME_CLIENT_HEADER "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/presentation-time-client-protocol.h")
set(OWUI_PRESENTATION_TIME_PRIVATE_CODE "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/presentation-time-protocol.c")
set(OWUI_XDG_FOREIGN_CLIENT_HEADER "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/xdg-foreign-unstable-v2-client-protocol.h")
set(OWUI_XDG_FOREIGN_PRIVATE_CODE "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}/xdg-foreign-unstable-v2-protocol.c")

add_custom_command(
	OUTPUT
		"${OWUI_XDG_SHELL_CLIENT_HEADER}"
		"${OWUI_XDG_SHELL_PRIVATE_CODE}"
		"${OWUI_XDG_DECORATION_CLIENT_HEADER}"
		"${OWUI_XDG_DECORATION_PRIVATE_CODE}"
		"${OWUI_PRESENTATION_TIME_CLIENT_HEADER}"
		"${OWUI_PRESENTATION_TIME_PRIVATE_CODE}"
		"${OWUI_XDG_FOREIGN_CLIENT_HEADER}"
		"${OWUI_XDG_FOREIGN_PRIVATE_CODE}"
	COMMAND "${CMAKE_COMMAND}" -E make_directory "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" client-header "${OWUI_XDG_SHELL_PROTOCOL}" "${OWUI_XDG_SHELL_CLIENT_HEADER}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" private-code "${OWUI_XDG_SHELL_PROTOCOL}" "${OWUI_XDG_SHELL_PRIVATE_CODE}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" client-header "${OWUI_XDG_DECORATION_PROTOCOL}" "${OWUI_XDG_DECORATION_CLIENT_HEADER}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" private-code "${OWUI_XDG_DECORATION_PROTOCOL}" "${OWUI_XDG_DECORATION_PRIVATE_CODE}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" client-header "${OWUI_PRESENTATION_TIME_PROTOCOL}" "${OWUI_PRESENTATION_TIME_CLIENT_HEADER}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" private-code "${OWUI_PRESENTATION_TIME_PROTOCOL}" "${OWUI_PRESENTATION_TIME_PRIVATE_CODE}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" client-header "${OWUI_XDG_FOREIGN_PROTOCOL}" "${OWUI_XDG_FOREIGN_CLIENT_HEADER}"
	COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" private-code "${OWUI_XDG_FOREIGN_PROTOCOL}" "${OWUI_XDG_FOREIGN_PRIVATE_CODE}"
	DEPENDS "${OWUI_XDG_SHELL_PROTOCOL}" "${OWUI_XDG_DECORATION_PROTOCOL}" "${OWUI_PRESENTATION_TIME_PROTOCOL}" "${OWUI_XDG_FOREIGN_PROTOCOL}"
	VERBATIM
)
set_source_files_properties(
	"${OWUI_XDG_SHELL_CLIENT_HEADER}"
	"${OWUI_XDG_SHELL_PRIVATE_CODE}"
	"${OWUI_XDG_DECORATION_CLIENT_HEADER}"
	"${OWUI_XDG_DECORATION_PRIVATE_CODE}"
	"${OWUI_PRESENTATION_TIME_CLIENT_HEADER}"
	"${OWUI_PRESENTATION_TIME_PRIVATE_CODE}"
	"${OWUI_XDG_FOREIGN_CLIENT_HEADER}"
	"${OWUI_XDG_FOREIGN_PRIVATE_CODE}"
	PROPERTIES GENERATED TRUE
)
add_custom_target(owui_backend_wayland_protocols DEPENDS
	"${OWUI_XDG_SHELL_CLIENT_HEADER}"
	"${OWUI_XDG_SHELL_PRIVATE_CODE}"
	"${OWUI_XDG_DECORATION_CLIENT_HEADER}"
	"${OWUI_XDG_DECORATION_PRIVATE_CODE}"
	"${OWUI_PRESENTATION_TIME_CLIENT_HEADER}"
	"${OWUI_PRESENTATION_TIME_PRIVATE_CODE}"
	"${OWUI_XDG_FOREIGN_CLIENT_HEADER}"
	"${OWUI_XDG_FOREIGN_PRIVATE_CODE}"
)

add_library(owui_backend_Wayland_GL3 INTERFACE)
target_sources(owui_backend_Wayland_GL3 INTERFACE
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Platform_Wayland.cpp"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Renderer_GL3.cpp"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Backend_Wayland_GL3.cpp"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Portal_Uri.cpp"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Platform_Wayland.h"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Portal_Uri.h"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Renderer_GL3.h"
	"${CMAKE_CURRENT_LIST_DIR}/OnlyWayUi_Include_GL3.h"
	"${OWUI_XDG_SHELL_CLIENT_HEADER}"
	"${OWUI_XDG_SHELL_PRIVATE_CODE}"
	"${OWUI_XDG_DECORATION_CLIENT_HEADER}"
	"${OWUI_XDG_DECORATION_PRIVATE_CODE}"
	"${OWUI_PRESENTATION_TIME_CLIENT_HEADER}"
	"${OWUI_PRESENTATION_TIME_PRIVATE_CODE}"
	"${OWUI_XDG_FOREIGN_CLIENT_HEADER}"
	"${OWUI_XDG_FOREIGN_PRIVATE_CODE}"
)
target_include_directories(owui_backend_Wayland_GL3 INTERFACE "${OWUI_WAYLAND_PROTOCOLS_OUTPUT_DIR}")
add_dependencies(owui_backend_Wayland_GL3 owui_backend_wayland_protocols)
target_link_libraries(owui_backend_Wayland_GL3 INTERFACE
	owui_backend_common_headers
	PkgConfig::OWUI_WAYLAND_CLIENT
	PkgConfig::OWUI_WAYLAND_CURSOR
	PkgConfig::OWUI_WAYLAND_EGL
	PkgConfig::OWUI_XKBCOMMON
	PkgConfig::OWUI_EGL
	PkgConfig::OWUI_DBUS
)

if(OWUI_IS_ROOT_PROJECT)
	install(DIRECTORY "./"
		DESTINATION "${CMAKE_INSTALL_DATADIR}/Backends"
	)
endif()
